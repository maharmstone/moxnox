# number tables, generated by build-machine tools (all take LIMB_BITS NAIL_BITS)
foreach(tool gen-fac gen-fib gen-bases gen-trialdivtab gen-psqr gen-jacobitab)
    add_custom_command(
        OUTPUT ${GEN}/tools/${tool}${HOST_EXECUTABLE_SUFFIX}
        COMMAND cc -O2 -std=gnu17 -o ${GEN}/tools/${tool}${HOST_EXECUTABLE_SUFFIX} ${CMAKE_CURRENT_SOURCE_DIR}/repo/${tool}.c
        DEPENDS repo/${tool}.c
        VERBATIM
    )
endforeach()

macro(gmp_table output tool args)
    add_custom_command(
        OUTPUT ${GEN}/${output}
        COMMAND sh -c "'${GEN}/tools/${tool}${HOST_EXECUTABLE_SUFFIX}' ${args} > '${GEN}/${output}'"
        DEPENDS ${GEN}/tools/${tool}${HOST_EXECUTABLE_SUFFIX}
        VERBATIM
    )
endmacro()

# each mpn operation compiles a generic source with OPERATION_<name> defined,
# via a generated wrapper (upstream uses symlinks plus -DOPERATION_<name>)
set(GMP_MPN_SOURCES)
macro(gmp_mpn_operation name src)
    file(CONFIGURE OUTPUT ${GEN}/mpn/${name}.c CONTENT "#define OPERATION_${name} 1\n#include \"${CMAKE_CURRENT_SOURCE_DIR}/repo/${src}\"\n" @ONLY)
    list(APPEND GMP_MPN_SOURCES ${GEN}/mpn/${name}.c)
endmacro()

    ${CMAKE_CURRENT_SOURCE_DIR}
    ${GEN}/mpn
    ${CMAKE_CURRENT_SOURCE_DIR}/repo
)

    ${USERSPACE_CFLAGS}
    -Wno-conversion
    -Wno-sign-conversion
    -Wno-unused-parameter
    -Wno-unused-but-set-variable
    -Wno-array-parameter
    -Wno-sign-compare
    -Wno-cast-function-type
    -Wno-implicit-fallthrough
    -Wno-missing-field-initializers
    -Wno-type-limits
    -Wno-empty-body
    -Wno-unused-function
    -Wno-unused-value
    -Wno-unused-variable
    $<$<C_COMPILER_ID:GNU>:-Wno-old-style-declaration>
    $<$<C_COMPILER_ID:Clang>:-Wno-null-pointer-subtraction -Wno-tautological-constant-out-of-range-compare>
)
